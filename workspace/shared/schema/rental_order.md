/******************************************/
rental_order,其中 vendor_id 表示 SaaS 的 tenant_id（即租户 id)
/******************************************/
CREATE TABLE `rental_order` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '主键id',
  `active` bit(1) DEFAULT b'1' COMMENT 'active',
  `deleted` bit(1) DEFAULT b'0' COMMENT 'deleted',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `creator_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '创建人id',
  `updater_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '更新人id',
  `order_id` bigint unsigned NOT NULL COMMENT '订单id',
  `order_code` varchar(30) NOT NULL COMMENT '订单编号',
  `store_id` bigint unsigned NOT NULL COMMENT '门店id',
  `vendor_id` bigint unsigned NOT NULL COMMENT '供应商id',
  `vendor_brand_id` bigint unsigned NOT NULL DEFAULT '1' COMMENT '供应商品牌id',
  `pickup_location_id` bigint unsigned NOT NULL COMMENT '取车门店id',
  `dropoff_location_id` bigint unsigned NOT NULL COMMENT '还车门店id',
  `pickup_time` datetime NOT NULL COMMENT '取车时间',
  `dropoff_time` datetime NOT NULL COMMENT '还车时间',
  `appoint_model_id` bigint unsigned NOT NULL COMMENT '预约车型id',
  `appoint_stock_id` bigint unsigned NOT NULL COMMENT '预约库存id',
  `appoint_days` tinyint unsigned DEFAULT NULL COMMENT '预约天数',
  `prepaid_amount` decimal(10,2) NOT NULL DEFAULT '0.00' COMMENT '预付金额',
  `pay_on_arrival_amount` decimal(10,2) NOT NULL DEFAULT '0.00' COMMENT '到付金额',
  `currency` varchar(10) DEFAULT NULL COMMENT '货币单位',
  `rate_version` bigint DEFAULT NULL COMMENT '汇率版本号，',
  `vendor_currency` varchar(10) DEFAULT NULL COMMENT '供应商本币',
  `instant_confirm` bit(1) DEFAULT b'1' COMMENT '是否是立即确认，true-是，false-否（二次确认），默认是',
  `confirm_duration` int unsigned DEFAULT NULL COMMENT '确认时长，单位默认小时',
  `confirm_time` datetime DEFAULT NULL COMMENT '确认时间',
  `confirmer_id` bigint unsigned DEFAULT NULL COMMENT '确认人id',
  `reject_time` datetime DEFAULT NULL COMMENT '拒单时间',
  `rejecter_id` bigint unsigned DEFAULT NULL COMMENT '拒单人id',
  `pending_expires_at` datetime DEFAULT NULL COMMENT '占位中超时时间（用于 PENDING 自动收敛）',
  `last_atp_checked_at` datetime DEFAULT NULL COMMENT '最近一次 ATP 可用库存校验时间',
  `reject_reason_code` varchar(50) DEFAULT NULL COMMENT '拒单原因编码：INSUFFICIENT_STOCK/PENDING_TIMEOUT/MANUAL_REJECT',
  `is_disputed` tinyint(1) NOT NULL DEFAULT '0' COMMENT '争议标记：0-否，1-是（非主状态）',
  `order_type` tinyint DEFAULT '1' COMMENT '订单履约类型：1-普通订单，2-online checkin，3-自主取还',
  `create_type` tinyint NOT NULL DEFAULT '2' COMMENT '订单创建方式 1:到店订单 2:api创建 3:手动导入 4:线下外部渠道订单（1和4都为线下订单）',
  `ota_insurance` varchar(20) DEFAULT NULL COMMENT '是否购买平台保险：yes-是，no-否',
  `enable_online_checkin` bit(1) DEFAULT b'0' COMMENT '供应商是否开启online checkin：0-未开启，1-开启',
  `status` tinyint NOT NULL DEFAULT '6' COMMENT '订单状态：-2-已拒绝，-1-已取消，0-占位中，1-已确认，2-已取车，3-已还车，4-已完成，5-No-show，6-已创建',
  `enable_self_pickup_dropoff` tinyint(1) DEFAULT NULL COMMENT '是否启用自助取还（供应商扩展配置快照）',
  `self_pickup_name` varchar(255) DEFAULT NULL COMMENT '自助取还取车位置',
  `self_return_name` varchar(255) DEFAULT NULL COMMENT '自助取还还车位置',
  `self_pickup_address` varchar(255) DEFAULT NULL COMMENT '自助取还取车地址',
  `self_return_address` varchar(255) DEFAULT NULL COMMENT '自助取还还车地址',
  `self_pickup_coordinate` varchar(255) DEFAULT NULL COMMENT '自助取还取车位置经度',
  `self_return_coordinate` varchar(255) DEFAULT NULL COMMENT '自助取还还车位置纬度',
  `gotar_adult` int DEFAULT NULL COMMENT 'Gotar 订单：搭乘成人人数',
  `gotar_child` int DEFAULT NULL COMMENT 'Gotar 订单：搭乘儿童人数',
  `gotar_infant` int DEFAULT NULL COMMENT 'Gotar 订单：搭乘婴儿人数',
  `gotar_driver_gender` varchar(8) DEFAULT NULL COMMENT 'Gotar 订单：驾驶员性别（M/F）',
  `time_zone` int DEFAULT NULL COMMENT '时区',
  PRIMARY KEY (`id`),
  KEY `idx_update_time` (`update_time`),
  KEY `idx_ro_order_id` (`order_id`)
) ENGINE=InnoDB AUTO_INCREMENT=24026 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='租车订单表'
;
