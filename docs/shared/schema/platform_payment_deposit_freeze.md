/******************************************/
platform_payment_deposit_freeze
/******************************************/
CREATE TABLE `platform_payment_deposit_freeze` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `active` bit(1) DEFAULT b'1' COMMENT 'active',
  `deleted` bit(1) DEFAULT b'0' COMMENT 'deleted',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT 'create_time in UTC',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'update_time in UTC',
  `creator_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '创建人id',
  `updater_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '更新人id',
  `deposit_id` bigint NOT NULL COMMENT '订单押金预授权总表id',
  `payment_intent_id` varchar(64) NOT NULL COMMENT 'pi_xxx 支付意图唯一标识',
  `client_secret` varchar(128) DEFAULT NULL COMMENT '用于前端 confirm',
  `latest_charge_id` varchar(64) DEFAULT NULL COMMENT 'charge_xxx（捕获后才有）',
  `amount` bigint NOT NULL COMMENT '预授权总金额，单位：该币种的最小货币单位（如USD为cents，JPY为yen）',
  `amount_captured` bigint NOT NULL DEFAULT '0' COMMENT '已实际扣款金额，单位：该币种的最小货币单位（如USD为cents，JPY为yen）',
  `currency` varchar(5) NOT NULL COMMENT '币种代码，如 USD, JPY, EUR',
  `card_brand` varchar(20) DEFAULT NULL COMMENT '卡品牌: visa, mastercard, american_express 等',
  `card_last4` varchar(6) DEFAULT NULL COMMENT '卡号后四位',
  `card_holder` varchar(60) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '持卡人姓名',
  `card_exp_month` tinyint unsigned DEFAULT NULL COMMENT '卡到期月 (1-12)',
  `card_exp_year` smallint unsigned DEFAULT NULL COMMENT '卡到期年 (如 2028)',
  `card_country` varchar(5) DEFAULT NULL COMMENT '发卡国家 ISO 代码，如 US, CN',
  `card_funding` varchar(20) DEFAULT NULL COMMENT '卡类型: credit, debit, prepaid, unknown 等',
  `status` tinyint NOT NULL COMMENT '1=待支付 2=待确认 3=待认证 4=待捕获 5=成功 6=已取消 7=已过期',
  `extended_authorization_enabled` tinyint(1) DEFAULT '0' COMMENT '是否已开通30天延长授权',
  `authorization_expires_at` datetime DEFAULT NULL COMMENT '预授权到期时间（7天或30天）',
  `last_increment_at` datetime DEFAULT NULL COMMENT '最近一次增量授权时间（用于14天方案）',
  `incremental_count` tinyint unsigned DEFAULT NULL COMMENT '增量授权次数',
  `expected_expires_at` datetime DEFAULT NULL COMMENT '业务预期最晚过期时间',
  `application_result` int NOT NULL COMMENT '冻结申请结果 1-成功，2-失败',
  `effect_status` int NOT NULL COMMENT '是否生效 0-否，1-是',
  PRIMARY KEY (`id`),
  UNIQUE KEY `payment_intent_id` (`payment_intent_id`),
  UNIQUE KEY `deposit_id_effect_status` (`deposit_id`,`effect_status`),
  KEY `idx_order_id` (`deposit_id`),
  KEY `idx_pi` (`payment_intent_id`),
  KEY `idx_status` (`status`),
  KEY `idx_expires` (`authorization_expires_at`),
  KEY `idx_deposit_last4` (`deposit_id`,`card_last4`),
  KEY `idx_card_brand` (`card_brand`)
) ENGINE=InnoDB AUTO_INCREMENT=17 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='预授权冻结明细表'
;
